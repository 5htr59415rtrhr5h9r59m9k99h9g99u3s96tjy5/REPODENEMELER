          const app = initializeApp(firebaseConfig);
          db = getFirestore(app);
          firebaseHazir = true;
        }
      } catch (err) {
        console.warn('Firebase başlatılamadı:', err);
      }

      let telegramAyar = { botToken: '', chatId: '', notifyOnVisit: false, notifyOnComment: false, notifyOnClick: false };

      async function telegramAyarYukle() {
        if (!firebaseHazir) return;
        try {
          const snap = await getDoc(doc(db, 'settings', 'telegram'));
          if (snap.exists()) {
            telegramAyar = { ...telegramAyar, ...snap.data() };
          }
        } catch (err) {
          console.warn('Telegram ayarları okunamadı:', err);
        }
        telegramHazirlaEtkinlikler();
      }

      async function telegramBildir(mesaj) {
        if (!telegramAyar.botToken || !telegramAyar.chatId) return;
        try {
          await fetch(`https://api.telegram.org/bot${telegramAyar.botToken}/sendMessage`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ chat_id: telegramAyar.chatId, text: mesaj, parse_mode: 'HTML' })
          });
        } catch (err) {
          console.warn('Telegram bildirimi gönderilemedi:', err);
        }
      }

      function telegramSaatDamgasi() {
        return new Date().toLocaleString('tr-TR', { timeZone: 'Europe/Istanbul' });
      }

      function telegramHazirlaEtkinlikler() {
        if (telegramAyar.notifyOnVisit && !sessionStorage.getItem('ziyaretBildirildi')) {
          sessionStorage.setItem('ziyaretBildirildi', '1');
          telegramBildir(`🌐 <b>Siteye yeni ziyaret</b>\n55fındık.com sitesine bir ziyaretçi girdi.\n🕒 ${telegramSaatDamgasi()}`);
        }

        if (telegramAyar.notifyOnClick) {
          document.querySelectorAll('.btn-contact, .konum-float-btn, .galeri-btn, .yorum-bak-btn, .yorum-gonder-btn, #tumunuGosterBtn').forEach((el) => {
            el.addEventListener('click', () => {
              let etiket = el.getAttribute('title') || el.textContent.trim() || 'Bir buton';
              const kart = el.closest('.contact-card');
              if (kart) {
                const isim = kart.querySelector('h3')?.textContent?.trim();
                if (isim) etiket = `${isim} — ${etiket}`;
              }
              if (el.id === 'konumFloatBtn') etiket = 'Konumumuz butonu';
              telegramBildir(`🖱️ <b>Tıklama oldu</b>\n${etiket}\n🕒 ${telegramSaatDamgasi()}`);
            });
          });
        }
      }

      function renderSlider() {
        if (!sliderTrack) return;
        sliderTrack.innerHTML = '';
        const kartlar = [...fotoList, ...fotoList];
        kartlar.forEach((foto, index) => {
          const card = document.createElement('div');
          card.className = 'galeri-card';
          const aciklama = foto.aciklama || `Alım Anı #${(index % fotoList.length) + 1}`;
          card.innerHTML = `
            <img src="${foto.url}" alt="${aciklama}" loading="lazy" referrerpolicy="no-referrer" />
            <span class="card-cap">${aciklama}</span>
          `;
          sliderTrack.appendChild(card);
        });
      }

      function renderModal(siralama) {
        if (!modalGrid) return;
        modalGrid.innerHTML = '';
        let sirali = [...fotoList];
        if (siralama === 'eski') {
          sirali.sort((a, b) => (a.createdAt || 0) - (b.createdAt || 0));
        } else {
          sirali.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
        }
        sirali.forEach((foto, i) => {
          const card = document.createElement('div');
          card.className = 'galeri-card';
          card.innerHTML = `<img src="${foto.url}" alt="${foto.aciklama || 'Foto ' + (i+1)}" loading="lazy" referrerpolicy="no-referrer" /><span class="card-cap">${foto.aciklama || '#' + (i+1)}</span>`;
          modalGrid.appendChild(card);
        });
      }

      function yildizGoster(puan) {
        let html = '';
        for (let i = 1; i <= 5; i++) {
          if (i <= puan) {
            html += '<i class="fas fa-star"></i>';
          } else {
            html += '<i class="fas fa-star bos"></i>';
          }
        }
        return html;
      }

      function yorumKartOlustur(isim, tel, metin, puan) {
        const kart = document.createElement('div');
        kart.className = 'yorum-kart';
        kart.innerHTML = `
          <div class="yorum-ust">
            <span class="yorum-isim">${isim || 'Anonim'}</span>
            <span class="yorum-tel">${tel || '---'}</span>
          </div>
          <div class="yorum-yildizlar">${yildizGoster(puan || 0)}</div>
          <div class="yorum-metin">${metin}</div>
        `;
        return kart;
      }

      function yorumlariCanliDinle() {
        const yorumListesi = document.getElementById('yorumListesi');
        if (!db || !yorumListesi) return;
        const q = query(collection(db, 'comments'), where('approved', '==', true));
        onSnapshot(q, (snap) => {
          tumYorumlar = snap.docs.map((d) => ({ id: d.id, ...d.data() }));
          tumYorumlar.sort((a, b) => {
            const ta = a.createdAt?.toMillis?.() || 0;
            const tb = b.createdAt?.toMillis?.() || 0;
            return tb - ta;
          });
          yorumListesi.innerHTML = '';
          tumYorumlar.slice(0, 3).forEach((y) => {
            yorumListesi.appendChild(yorumKartOlustur(y.name, y.phone, y.text, y.rating));
          });
          renderYorumModal();
        }, (err) => console.warn('Yorumlar okunamadı:', err));
      }

      function renderYorumModal() {
        const modalListe = document.getElementById('yorumModalListesi');
        if (!modalListe) return;
        const siralama = document.getElementById('yorumSiralama')?.value || 'tarih-yeni';
        let sirali = [...tumYorumlar];
        switch (siralama) {
          case 'tarih-yeni':
            sirali.sort((a, b) => (b.createdAt?.toMillis?.() || 0) - (a.createdAt?.toMillis?.() || 0));
            break;
          case 'tarih-eski':
            sirali.sort((a, b) => (a.createdAt?.toMillis?.() || 0) - (b.createdAt?.toMillis?.() || 0));
            break;
          case 'yildiz-azalan':
            sirali.sort((a, b) => (b.rating || 0) - (a.rating || 0));
            break;
          case 'yildiz-artan':
            sirali.sort((a, b) => (a.rating || 0) - (b.rating || 0));
            break;
        }
        modalListe.innerHTML = '';
        if (sirali.length === 0) {
          modalListe.innerHTML = '<p style="color:var(--krem-dim); text-align:center;">Henüz onaylanmış yorum yok.</p>';
          return;
        }
        sirali.forEach((y) => {
          modalListe.appendChild(yorumKartOlustur(y.name, y.phone, y.text, y.rating));
        });
      }

      function galeriCanliDinle() {
        if (!db) return;
        const q = query(collection(db, 'gallery'));
        onSnapshot(q, (snap) => {
          if (!snap.empty) {
            const docsSirali = snap.docs.slice().sort((a, b) => {
              const ta = a.data().createdAt?.toMillis?.() || 0;
              const tb = b.data().createdAt?.toMillis?.() || 0;
              return tb - ta;
            });
            const calisanlar = docsSirali.filter((d) => {
              const url = d.data().url || '';
              return url.includes('i.ibb.co') || url.includes('i.postimg.cc');
            });
            if (calisanlar.length > 0) {
              fotoList = calisanlar.map((d) => ({
                url: d.data().url,
                aciklama: d.data().aciklama || '',
                createdAt: d.data().createdAt?.toMillis?.() || Date.now()
              }));
            }
          }
          renderSlider();
          renderModal(document.getElementById('fotoSiralama')?.value || 'yeni');
        }, (err) => console.warn('Galeri okunamadı:', err));
      }

      let secilenYildiz = 0;
      const yildizSecici = document.getElementById('yildizSecici');
      if (yildizSecici) {
        const yildizlar = yildizSecici.querySelectorAll('.star');
        yildizlar.forEach((yildiz) => {
          yildiz.addEventListener('click', () => {
            secilenYildiz = parseInt(yildiz.dataset.deger);
            yildizlar.forEach((y) => {
              if (parseInt(y.dataset.deger) <= secilenYildiz) {
                y.classList.add('aktif');
              } else {
                y.classList.remove('aktif');
              }
            });
          });
        });
      }

      const yorumDurum = document.getElementById('yorumDurum');
      const gonderBtn = document.getElementById('yorumGonderBtn');
      if (gonderBtn) {
        gonderBtn.addEventListener('click', async () => {
          const isim = document.getElementById('yorumIsim').value.trim();
          const tel = document.getElementById('yorumTel').value.trim();
          const metin = document.getElementById('yorumMetin').value.trim();
          if (!isim || !tel || !metin) {
            alert('Lütfen isim, telefon ve yorum alanlarını doldurun.');
            return;
          }
          if (secilenYildiz === 0) {
            alert('Lütfen yıldız verin.');
            return;
          }
          if (firebaseHazir) {
            try {
              await addDoc(collection(db, 'comments'), {
                name: isim, phone: tel, text: metin,
                approved: false, rating: secilenYildiz, createdAt: serverTimestamp()
              });
              if (yorumDurum) {
                yorumDurum.classList.remove('hata');
                yorumDurum.textContent = 'Teşekkürler! Yorumunuz incelendikten sonra sitede yayınlanacak.';
              }
              if (telegramAyar.notifyOnComment) {
                telegramBildir(`💬 <b>Yeni yorum geldi</b>\n👤 ${isim}\n📞 ${tel}\n⭐ ${secilenYildiz}/5\n📝 ${metin}\n🕒 ${telegramSaatDamgasi()}`);
              }
            } catch (err) {
              console.error(err);
              if (yorumDurum) {
                yorumDurum.classList.add('hata');
                yorumDurum.textContent = 'Yorum gönderilemedi.';
              }
            }
          }
          document.getElementById('yorumIsim').value = '';
          document.getElementById('yorumTel').value = '';
          document.getElementById('yorumMetin').value = '';
          secilenYildiz = 0;
          if (yildizSecici) {
            yildizSecici.querySelectorAll('.star').forEach((y) => y.classList.remove('aktif'));
          }
        });
      }

      const modalOverlay = document.getElementById('modalOverlay');
      const modalCloseBtn = document.getElementById('modalCloseBtn');
      const tumunuGosterBtn = document.getElementById('tumunuGosterBtn');
      const fotoSiralama = document.getElementById('fotoSiralama');

      if (tumunuGosterBtn) tumunuGosterBtn.addEventListener('click', () => {
        renderModal(fotoSiralama?.value || 'yeni');
        modalOverlay.classList.add('show');
        document.body.style.overflow = 'hidden';
      });
      if (modalCloseBtn) modalCloseBtn.addEventListener('click', () => {
        modalOverlay.classList.remove('show');
        document.body.style.overflow = '';
      });
      if (modalOverlay) modalOverlay.addEventListener('click', (e) => {
        if (e.target === modalOverlay) {
          modalOverlay.classList.remove('show');
          document.body.style.overflow = '';
        }
      });
      if (fotoSiralama) fotoSiralama.addEventListener('change', () => {
        renderModal(fotoSiralama.value);
      });

      const yorumModalOverlay = document.getElementById('yorumModalOverlay');
      const yorumModalCloseBtn = document.getElementById('yorumModalCloseBtn');
      const yorumBakBtn = document.getElementById('yorumBakBtn');
      if (yorumBakBtn) yorumBakBtn.addEventListener('click', () => {
        renderYorumModal();
        yorumModalOverlay.classList.add('show');
        document.body.style.overflow = 'hidden';
      });
      if (yorumModalCloseBtn) yorumModalCloseBtn.addEventListener('click', () => {
        yorumModalOverlay.classList.remove('show');
        document.body.style.overflow = '';
      });
      if (yorumModalOverlay) yorumModalOverlay.addEventListener('click', (e) => {
        if (e.target === yorumModalOverlay) {
          yorumModalOverlay.classList.remove('show');
          document.body.style.overflow = '';
        }
      });
      const siralamaSelect = document.getElementById('yorumSiralama');
      if (siralamaSelect) siralamaSelect.addEventListener('change', renderYorumModal);

      document.addEventListener('keydown', (e) => {
        if (e.key === 'Escape') {
          if (modalOverlay.classList.contains('show')) {
            modalOverlay.classList.remove('show');
            document.body.style.overflow = '';
          }
          if (yorumModalOverlay.classList.contains('show')) {
            yorumModalOverlay.classList.remove('show');
            document.body.style.overflow = '';
          }
        }
      });

      const konumBtn = document.getElementById('konumFloatBtn');
      if (konumBtn) {
        konumBtn.addEventListener('click', (e) => {
          e.preventDefault();
          const hedef = document.getElementById('harita');
          if (hedef) hedef.scrollIntoView({ behavior: 'smooth', block: 'start' });
        });
      }

      window.addEventListener('DOMContentLoaded', () => {
        document.getElementById('yil').textContent = new Date().getFullYear();
        renderSlider();
        renderModal('yeni');
        if (firebaseHazir) {
          galeriCanliDinle();
          yorumlariCanliDinle();
          telegramAyarYukle();
        } else {
          telegramHazirlaEtkinlikler();
        }
      });
    })();
  </script>
